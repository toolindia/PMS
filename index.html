// ========================================== // OFBUZZ PMS - MULTI-MODULE BACKEND ENGINE // ========================================== function ensureDatabaseSchema() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const schema = { "Settings": ["PropertyKey", "PropertyValue"], "Rooms": ["RoomNo", "RoomType", "Price", "Status"], "Bookings": ["BookingId", "RoomNo", "GuestName", "Phone", "CheckIn", "CheckOut", "Advance", "IdPhotoUrl", "Status"], "MenuItems": ["ItemId", "Category", "ItemName", "Price", "Available"], "FoodOrders": ["OrderId", "RoomNo", "BookingId", "Items", "Amount", "Timestamp"], "Staff": ["StaffId", "FullName", "Role", "Phone", "ShiftStatus"], "Attendance": ["RecordId", "StaffId", "StaffName", "Action", "Timestamp"], "Invoices": ["InvoiceId", "BookingId", "RoomNo", "GuestName", "RoomTotal", "FoodTotal", "AdvancePaid", "FinalPaid", "PaymentMode", "Timestamp"], "Feedback": ["FeedbackId", "RoomNo", "GuestName", "Cleanliness", "Staff", "Amenities", "Food", "Comments", "Timestamp"] }; for (const sheetName in schema) { let sheet = ss.getSheetByName(sheetName); const headers = schema[sheetName]; if (!sheet) { sheet = ss.insertSheet(sheetName); } if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold"); sheet.setFrozenRows(1); if (sheetName === "Settings") { sheet.appendRow(["PropertyName", "Grand Heritage Residency"]); sheet.appendRow(["AdminPin", "1234"]); sheet.appendRow(["ReceptionPin", "0000"]); // Default Shop ID for Reception sheet.appendRow(["Currency", "INR"]); } if (sheetName === "MenuItems") { sheet.appendRow(["M101", "Beverages", "Masala Tea", 30, "Yes"]); sheet.appendRow(["M102", "Beverages", "Filter Coffee", 50, "Yes"]); sheet.appendRow(["M103", "Breakfast", "Club Sandwich", 120, "Yes"]); sheet.appendRow(["M104", "Main Course", "Paneer Butter Masala", 260, "Yes"]); sheet.appendRow(["M105", "Breads", "Butter Naan", 40, "Yes"]); } if (sheetName === "Rooms") { sheet.appendRow(["101", "Deluxe Room", 2500, "Vacant"]); sheet.appendRow(["102", "Deluxe Room", 2500, "Vacant"]); sheet.appendRow(["103", "Executive Suite", 4000, "Vacant"]); sheet.appendRow(["104", "Standard Room", 1800, "Vacant"]); } if (sheetName === "Staff") { sheet.appendRow(["ST-01", "Ramesh Kumar", "Front Desk", "9876500001", "On Duty"]); sheet.appendRow(["ST-02", "Sunil Verma", "Housekeeping", "9876500002", "Off Duty"]); sheet.appendRow(["ST-03", "Anil Sharma", "Kitchen Service", "9876500003", "On Duty"]); } } } const defaultSheet1 = ss.getSheetByName("Sheet1"); if (defaultSheet1 && ss.getSheets().length > 1) { try { ss.deleteSheet(defaultSheet1); } catch (e) {} } } function doGet(e) { try { ensureDatabaseSchema(); const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Initial Property Profile & Settings if (action === "getPropertyConfig") { const settings = getSettingsMap(ss); return createJsonResponse({ success: true, propertyName: settings["PropertyName"] || "Hotel Portal", currency: settings["Currency"] || "INR" }); } // 2. Reception Desk Data (WITH PIN VERIFICATION) if (action === "getReceptionData") { const settings = getSettingsMap(ss); const clientPin = (e.parameter.pin || "").toString().trim(); const actualPin = (settings["ReceptionPin"] || "0000").toString().trim(); if (clientPin !== actualPin) { return createJsonResponse({ success: false, error: "Access verification code incorrect" }); } const rooms = getSheetDataAsJson(ss.getSheetByName("Rooms")); const activeBookings = getSheetDataAsJson(ss.getSheetByName("Bookings")).filter(b => b.Status === "Active"); const foodOrders = getSheetDataAsJson(ss.getSheetByName("FoodOrders")); return createJsonResponse({ success: true, propertyName: settings["PropertyName"] || "Hotel Operations", rooms: rooms, activeBookings: activeBookings, foodOrders: foodOrders }); } // 3. Dining & Restaurant Data if (action === "getDiningData") { const settings = getSettingsMap(ss); const menu = getSheetDataAsJson(ss.getSheetByName("MenuItems")); const activeBookings = getSheetDataAsJson(ss.getSheetByName("Bookings")).filter(b => b.Status === "Active"); const recentOrders = getSheetDataAsJson(ss.getSheetByName("FoodOrders")).reverse().slice(0, 20); return createJsonResponse({ success: true, propertyName: settings["PropertyName"] || "Hotel Operations", menu: menu, activeRooms: activeBookings.map(b => ({ RoomNo: b.RoomNo, GuestName: b.GuestName, BookingId: b.BookingId })), recentOrders: recentOrders }); } // 4. Staff Management Data if (action === "getStaffData") { const settings = getSettingsMap(ss); const staffList = getSheetDataAsJson(ss.getSheetByName("Staff")); const attendance = getSheetDataAsJson(ss.getSheetByName("Attendance")).reverse().slice(0, 30); return createJsonResponse({ success: true, propertyName: settings["PropertyName"] || "Hotel Operations", staffList: staffList, attendance: attendance }); } // 5. Admin / Analytics Data if (action === "getAdminData") { const settings = getSettingsMap(ss); const clientPin = (e.parameter.pin || "").toString().trim(); const actualPin = (settings["AdminPin"] || "1234").toString().trim(); if (clientPin !== actualPin) { return createJsonResponse({ success: false, error: "Access verification code incorrect" }); } const rooms = getSheetDataAsJson(ss.getSheetByName("Rooms")); const bookings = getSheetDataAsJson(ss.getSheetByName("Bookings")); const invoices = getSheetDataAsJson(ss.getSheetByName("Invoices")); const feedback = getSheetDataAsJson(ss.getSheetByName("Feedback")); const totalRooms = rooms.length; const occupiedRooms = rooms.filter(r => r.Status === "Occupied").length; const cleaningRooms = rooms.filter(r => r.Status === "Cleaning").length; const vacantRooms = totalRooms - (occupiedRooms + cleaningRooms); let totalRoomRev = 0, totalFoodRev = 0, grandTotalRev = 0; invoices.forEach(inv => { totalRoomRev += Number(inv.RoomTotal) || 0; totalFoodRev += Number(inv.FoodTotal) || 0; grandTotalRev += Number(inv.FinalPaid) || 0; }); return createJsonResponse({ success: true, propertyName: settings["PropertyName"] || "Hotel Operations", summary: { totalRooms: totalRooms, occupiedRooms: occupiedRooms, vacantRooms: vacantRooms, cleaningRooms: cleaningRooms, occupancyRate: totalRooms > 0 ? Math.round((occupiedRooms / totalRooms) * 100) + "%" : "0%", totalRoomRev: totalRoomRev, totalFoodRev: totalFoodRev, grandTotalRev: grandTotalRev }, invoices: invoices.reverse(), recentBookings: bookings.reverse().slice(0, 50), feedbacks: feedback.reverse() }); } return createJsonResponse({ success: false, error: "Requested routine not recognized" }); } catch (err) { return createJsonResponse({ success: false, error: err.toString() }); } } function doPost(e) { try { ensureDatabaseSchema(); const postData = JSON.parse(e.postData.contents); const action = postData.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); const timeNow = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm"); // Check-In if (action === "checkIn") { const roomNo = postData.roomNo.toString().trim(); const guestName = postData.guestName.trim(); const phone = postData.phone.trim(); const advance = Number(postData.advance) || 0; const photoDataUrl = postData.idPhotoBase64 || ""; const bookingId = "BK-" + new Date().getTime().toString().slice(-6); const bookingsSheet = ss.getSheetByName("Bookings"); bookingsSheet.appendRow([bookingId, roomNo, guestName, phone, timeNow, "", advance, photoDataUrl, "Active"]); updateRoomStatus(ss, roomNo, "Occupied"); return createJsonResponse({ success: true, bookingId: bookingId, message: "Guest check-in registered" }); } // Food Order Placement if (action === "addFoodOrder") { const roomNo = postData.roomNo.toString().trim(); const bookingId = postData.bookingId || ""; const items = postData.items.trim(); const amount = Number(postData.amount) || 0; const orderId = "ORD-" + new Date().getTime().toString().slice(-5); const foodSheet = ss.getSheetByName("FoodOrders"); foodSheet.appendRow([orderId, roomNo, bookingId, items, amount, timeNow]); return createJsonResponse({ success: true, orderId: orderId, message: "Dining order billed" }); } // Check-Out & Settlement if (action === "checkOut") { const bookingId = postData.bookingId; const roomNo = postData.roomNo.toString().trim(); const paymentMode = postData.paymentMode || "UPI"; const bookingsSheet = ss.getSheetByName("Bookings"); const bookingsData = bookingsSheet.getDataRange().getValues(); let rowIndex = -1, guestName = "", advancePaid = 0; for (let i = 1; i < bookingsData.length; i++) { if (bookingsData[i][0] == bookingId && bookingsData[i][8] == "Active") { rowIndex = i + 1; guestName = bookingsData[i][2]; advancePaid = Number(bookingsData[i][6]) || 0; break; } } if (rowIndex === -1) return createJsonResponse({ success: false, error: "Active record not located" }); const foodSheet = ss.getSheetByName("FoodOrders"); const foodData = foodSheet.getDataRange().getValues(); let foodTotal = 0; for (let j = 1; j < foodData.length; j++) { if (foodData[j][2] == bookingId) foodTotal += Number(foodData[j][4]) || 0; } const roomsSheet = ss.getSheetByName("Rooms"); const roomsData = roomsSheet.getDataRange().getValues(); let roomRent = 0; for (let k = 1; k < roomsData.length; k++) { if (roomsData[k][0].toString() == roomNo) { roomRent = Number(roomsData[k][2]) || 0; break; } } const netPayable = (roomRent + foodTotal) - advancePaid; const invoiceId = "INV-" + new Date().getTime().toString().slice(-6); bookingsSheet.getRange(rowIndex, 6).setValue(timeNow); bookingsSheet.getRange(rowIndex, 9).setValue("CheckedOut"); const invSheet = ss.getSheetByName("Invoices"); invSheet.appendRow([invoiceId, bookingId, roomNo, guestName, roomRent, foodTotal, advancePaid, netPayable, paymentMode, timeNow]); updateRoomStatus(ss, roomNo, "Cleaning"); return createJsonResponse({ success: true, invoiceId: invoiceId, netPayable: netPayable, message: "Settlement recorded" }); } // Mark Room Cleaned if (action === "markRoomCleaned") { const roomNo = postData.roomNo.toString().trim(); updateRoomStatus(ss, roomNo, "Vacant"); return createJsonResponse({ success: true, message: "Room marked vacant" }); } // Record Staff Attendance if (action === "recordAttendance") { const staffId = postData.staffId; const staffName = postData.staffName; const dutyAction = postData.dutyAction; // 'Check In' or 'Check Out' const recordId = "ATT-" + new Date().getTime().toString().slice(-5); ss.getSheetByName("Attendance").appendRow([recordId, staffId, staffName, dutyAction, timeNow]); // Update staff shift status const staffSheet = ss.getSheetByName("Staff"); const staffData = staffSheet.getDataRange().getValues(); for (let i = 1; i < staffData.length; i++) { if (staffData[i][0] == staffId) { staffSheet.getRange(i + 1, 5).setValue(dutyAction === "Check In" ? "On Duty" : "Off Duty"); break; } } return createJsonResponse({ success: true, message: "Attendance status updated" }); } // Submit Feedback if (action === "submitFeedback") { const feedbackId = "FB-" + new Date().getTime().toString().slice(-5); ss.getSheetByName("Feedback").appendRow([ feedbackId, postData.roomNo || "N/A", postData.guestName || "Guest", postData.cleanliness || 5, postData.staff || 5, postData.amenities || 5, postData.food || 5, postData.comments || "", timeNow ]); return createJsonResponse({ success: true, message: "Evaluation saved" }); } return createJsonResponse({ success: false, error: "Action routine undefined" }); } catch (err) { return createJsonResponse({ success: false, error: err.toString() }); } } // Helpers function getSettingsMap(ss) { const sheet = ss.getSheetByName("Settings"); const data = sheet.getDataRange().getValues(); const map = {}; for (let i = 1; i < data.length; i++) { if (data[i][0]) map[data[i][0].toString().trim()] = data[i][1]; } return map; } function updateRoomStatus(ss, roomNo, status) { const sheet = ss.getSheetByName("Rooms"); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0].toString() == roomNo.toString()) { sheet.getRange(i + 1, 4).setValue(status); break; } } } function getSheetDataAsJson(sheet) { if (!sheet) return []; const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; const headers = rows[0]; const out = []; for (let i = 1; i < rows.length; i++) { const row = rows[i], obj = {}; for (let j = 0; j < headers.length; j++) obj[headers[j]] = row[j]; out.push(obj); } return out; } function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
